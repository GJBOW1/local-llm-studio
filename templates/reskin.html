Here's a tiny useDebounce hook with proper cleanup:

typescript
function useDebounce<T>(value: T, ms = 300) {
  const [v, setV] = useState(value);
  useEffect(() => {
    const id = setTimeout(() => setV(value), ms);
    return () => clearTimeout(id);
  }, [value, ms]);
  return v;
}

It resets the timer on every keystroke, so v only updates once input settles. Drop it straight into your search box — no library needed.
